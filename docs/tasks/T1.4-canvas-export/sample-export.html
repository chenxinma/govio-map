-- 月度销售额汇总（口径：已支付订单）
SELECT strftime(o.created_at, '%Y-%m') AS 月份,
       c.区域, SUM(o.amount) AS 销售额
FROM orders o JOIN customers c ON o.customer_id = c.id
WHERE o.status = 'paid' AND o.amount < 100000
GROUP BY 1, 2
ORDER BY 1 DESC
